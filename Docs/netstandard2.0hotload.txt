        AppDomain domain = AppDomain.CreateDomain("PluginDomain");

        // 2. Instantiate a proxy object inside the new domain
        // Note: Proxy class must inherit from MarshalByRefObject
        Type proxyType = typeof(AssemblyProxy);
        AssemblyProxy proxy = (AssemblyProxy)domain.CreateInstanceAndUnwrap(
            proxyType.Assembly.FullName, 
            proxyType.FullName
        );

        // 3. Load and execute via proxy
        proxy.LoadAndRun("MyLegacyPlugin.dll");

        // 4. Destroy the domain (forces all contained assemblies to unload)
        AppDomain.Unload(domain);
    }
}
